# RETO 2 - SEDE CENTRAL SEGMENTADA CON FORTIGATE
# Config CLI lista para copiar/pegar en la consola de FortiOS
# Grupo 2 - Curso Telecomunicaciones UMG Cobán
##############################################################
# INSTRUCCIONES DE USO:
# 1. Conectate a la consola del FortiGate (Console tab en GNS3
#    o SSH una vez tenga IP).
# 2. Iniciá sesión (admin / sin contraseña la primera vez).
# 3. Pegá los bloques en orden. Cada "config...end" es
#    independiente, podés pegarlos uno por uno si preferís
#    ir verificando.
# 4. Al final, hacé backup: ver sección 8.
##############################################################


# ============================================================
# 1. HOSTNAME Y AJUSTES BÁSICOS
# ============================================================
config system global
    set hostname "FGT-SEDE-CENTRAL-G2"
    set admintimeout 60
end


# ============================================================
# 2. INTERFACES
#    Ajustar nombres de puerto (port1/port2/port3) según cómo
#    los reconozca tu FortiGate VM en GNS3 (revisá con
#    "get system interface" si no coinciden).
# ============================================================

# --- WAN / Tránsito hacia el núcleo (Grupo 1) ---
config system interface
    edit "port1"
        set alias "WAN-TRANSIT"
        set mode static
        set ip 10.255.0.2 255.255.255.0
        set allowaccess ping
        set role wan
    next
end

# --- LAN Corporativa ---
config system interface
    edit "port2"
        set alias "LAN-CORP"
        set mode static
        set ip 10.20.10.1 255.255.255.0
        set allowaccess ping https ssh
        set role lan
    next
end

# --- LAN Invitados ---
config system interface
    edit "port3"
        set alias "LAN-GUEST"
        set mode static
        set ip 10.20.20.1 255.255.255.0
        set allowaccess ping
        set role lan
    next
end


# ============================================================
# 3. RUTA ESTÁTICA POR DEFECTO (hacia el núcleo, Grupo 1)
# ============================================================
config router static
    edit 1
        set dst 0.0.0.0 0.0.0.0
        set gateway 10.255.0.1
        set device "port1"
    next
end


# ============================================================
# 4. DHCP - LAN CORPORATIVA
# ============================================================
config system dhcp server
    edit 1
        set dns-service default
        set default-gateway 10.20.10.1
        set netmask 255.255.255.0
        set interface "port2"
        config ip-range
            edit 1
                set start-ip 10.20.10.100
                set end-ip 10.20.10.200
            next
        end
    next
end


# ============================================================
# 5. DHCP - LAN INVITADOS
# ============================================================
config system dhcp server
    edit 2
        set dns-service default
        set default-gateway 10.20.20.1
        set netmask 255.255.255.0
        set interface "port3"
        config ip-range
            edit 1
                set start-ip 10.20.20.100
                set end-ip 10.20.20.200
            next
        end
    next
end


# ============================================================
# 6. OBJETOS DE DIRECCIÓN
#    VIP publicada por el Grupo 4 en la red de tránsito
#    (según el documento: 10.255.0.40, servicio HTTP/HTTPS)
# ============================================================
config firewall address
    edit "DMZ-SERVICIO-G4"
        set subnet 10.255.0.40 255.255.255.255
    next
    edit "LAN-CORP-NET"
        set subnet 10.20.10.0 255.255.255.0
    next
    edit "LAN-GUEST-NET"
        set subnet 10.20.20.0 255.255.255.0
    next
end

config firewall service custom
    edit "HTTP-HTTPS"
        set tcp-portrange 80 443
    next
end


# ============================================================
# 7. POLÍTICAS DE FIREWALL
#    Orden importa: las más específicas primero.
# ============================================================

# --- Policy 1: Corporativa -> Servicio DMZ del Grupo 4 ---
config firewall policy
    edit 1
        set name "CORP-to-DMZ-G4"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "LAN-CORP-NET"
        set dstaddr "DMZ-SERVICIO-G4"
        set action accept
        set schedule "always"
        set service "HTTP-HTTPS"
        set logtraffic all
        set nat enable
    next
end

# --- Policy 2: Corporativa -> Internet ---
config firewall policy
    edit 2
        set name "CORP-to-Internet"
        set srcintf "port2"
        set dstintf "port1"
        set srcaddr "LAN-CORP-NET"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set logtraffic all
        set nat enable
    next
end

# --- Policy 3: Invitados -> Internet (sin acceso a Corp) ---
config firewall policy
    edit 3
        set name "GUEST-to-Internet"
        set srcintf "port3"
        set dstintf "port1"
        set srcaddr "LAN-GUEST-NET"
        set dstaddr "all"
        set action accept
        set schedule "always"
        set service "ALL"
        set logtraffic all
        set nat enable
    next
end

# --- Policy 4: DENY explícito Invitados -> Corporativa ---
#     (evidencia clara del bloqueo para el reto; el deny
#     implícito de FortiOS ya lo haría, pero esta regla
#     deja el log más fácil de mostrar)
config firewall policy
    edit 4
        set name "DENY-GUEST-to-CORP"
        set srcintf "port3"
        set dstintf "port2"
        set srcaddr "LAN-GUEST-NET"
        set dstaddr "LAN-CORP-NET"
        set action deny
        set schedule "always"
        set service "ALL"
        set logtraffic all
    next
end


# ============================================================
# 8. RESTRICCIÓN DE ADMINISTRACIÓN
#    Solo la red corporativa puede administrar el FortiGate.
#    Quitamos https/ssh de port3 (ya no estaba habilitado) y
#    reforzamos con trusted hosts en la cuenta admin.
# ============================================================
config system admin
    edit "admin"
        set trusthost1 10.20.10.0 255.255.255.0
    next
end


# ============================================================
# 9. SYSLOG HACIA EL NOC/SOC (Grupo 5)
#    Confirmar puerto/formato exacto con Grupo 5 antes de
#    la demo; esto usa UDP 514 por defecto.
# ============================================================
config log syslogd setting
    set status enable
    set server "10.50.10.10"
    set port 514
    set facility local7
end

config log syslogd filter
    set severity information
end
